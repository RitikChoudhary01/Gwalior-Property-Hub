   133→// ============================================================
   134→// Schedule-visit dialog (team schedules on behalf of a lead)
   135→// ============================================================
   136→
   137→function ScheduleVisitDialog({
   138→  lead,
   139→  onClose,
   140→  onDone,
   141→}: {
   142→  lead: LeadDTO;
   143→  onClose: () => void;
   144→  onDone: () => void;
   145→}) {
   146→  const tomorrow = new Date(Date.now() + 86400000).toISOString().slice(0, 10);
   147→  const [date, setDate] = useState(tomorrow);
   148→  const [time, setTime] = useState("11:00 AM");
   149→  const [notes, setNotes] = useState("");
   150→  const [busy, setBusy] = useState(false);
   151→  const [error, setError] = useState<string | null>(null);
   152→
   153→  async function submit() {
   154→    if (!date || !time) {
   155→      setError("Pick a date and time slot for the visit.");
   156→      return;
   157→    }
   158→    setBusy(true);
   159→    setError(null);
   160→    try {
   161→      await api("/api/site-visits", {
   162→        method: "POST",
   163→        body: JSON.stringify({
   164→          propertyId: lead.property.id,
   165→          leadId: lead.id,
   166→          contactName: lead.name,
   167→          contactPhone: lead.phone,
   168→          preferredDate: date,
   169→          preferredTime: time,
   170→          notes: notes.trim() || undefined,
   171→        }),
   172→      });
   173→      toast.success("Site visit scheduled", {
   174→        description: `${lead.name} · ${formatDate(date)} at ${time}`,
   175→      });
   176→      onClose();
   177→      onDone();
   178→    } catch (err) {
   179→      setError(err instanceof ApiError ? err.message : "Could not schedule the visit.");
   180→    } finally {
   181→      setBusy(false);
   182→    }
   183→  }
   184→
   185→  return (
   186→    <Dialog open onOpenChange={(open) => !open && onClose()}>
   187→      <DialogContent className="sm:max-w-md">
   188→        <DialogHeader>
   189→          <DialogTitle>Schedule site visit</DialogTitle>
   190→          <DialogDescription>
   191→            Arrange a visit for {lead.name} to “{lead.property.title}” ({lead.property.locality},{" "}
   192→            {lead.property.city}).
   193→          </DialogDescription>
   194→        </DialogHeader>
   195→        <div className="grid gap-4">
   196→          <div className="grid gap-2">
   197→            <Label htmlFor="sv-date">Date</Label>
   198→            <Input
   199→              id="sv-date"
   200→              type="date"
   201→              value={date}
   202→              min={new Date().toISOString().slice(0, 10)}
   203→              onChange={(e) => {
   204→                setDate(e.target.value);
   205→                setError(null);
   206→              }}
   207→            />
   208→          </div>
   209→          <div className="grid gap-2">
   210→            <Label htmlFor="sv-time">Time slot</Label>
   211→            <Select value={time} onValueChange={setTime}>
   212→              <SelectTrigger id="sv-time" className="w-full">
   213→                <SelectValue />
   214→              </SelectTrigger>
   215→              <SelectContent>
   216→                {VISIT_TIME_SLOTS.map((slot) => (
   217→                  <SelectItem key={slot} value={slot}>
   218→                    {slot}
   219→                  </SelectItem>
   220→                ))}
   221→              </SelectContent>
   222→            </Select>
   223→          </div>
   224→          <div className="grid gap-2">
   225→            <Label htmlFor="sv-notes">Notes (optional)</Label>
   226→            <Textarea
   227→              id="sv-notes"
   228→              value={notes}
   229→              onChange={(e) => setNotes(e.target.value)}
   230→              placeholder="Anything the listing owner should prepare…"
   231→              rows={2}
   232→            />
   233→          </div>
   234→          {error ? <p className="text-sm text-destructive">{error}</p> : null}
   235→        </div>
   236→        <DialogFooter>
   237→          <Button variant="outline" onClick={onClose}>
   238→            Cancel
   239→          </Button>
   240→          <Button onClick={submit} disabled={busy}>
   241→            {busy ? <Loader2 className="size-4 animate-spin" aria-hidden="true" /> : <CalendarClock className="size-4" aria-hidden="true" />}
   242→            Schedule visit
   243→          </Button>
   244→        </DialogFooter>
   245→      </DialogContent>
   246→    </Dialog>
   247→  );
   248→}

   269→  const navigate = useAppStore((s) => s.navigate);
   270→  const [detail, setDetail] = useState<LeadDetail | null>(null);
   271→  const [loading, setLoading] = useState(true);
   272→  const [error, setError] = useState<string | null>(null);
   273→  const [note, setNote] = useState("");
   274→  const [followUp, setFollowUp] = useState("");
   275→  const [busy, setBusy] = useState(false);
   276→  const [scheduling, setScheduling] = useState(false);
   277→
   278→  const load = useCallback(async () => {
   279→    setLoading(true);
   280→    setError(null);
   281→    try {
   282→      const res = await api<LeadDetail>(`/api/leads/${leadId}`);
   283→      setDetail(res);
   284→      setFollowUp(res.lead.followUpDate ? res.lead.followUpDate.slice(0, 10) : "");
   285→    } catch (err) {
   286→      setError(err instanceof ApiError ? err.message : "Could not load this lead.");
   287→    } finally {
   288→      setLoading(false);
   289→    }
   290→  }, [leadId]);
   291→
   292→  useEffect(() => {
   293→    void load();
   294→  }, [load]);

   301→          method: "PATCH",
   302→          body: JSON.stringify(body),
   303→        }),
   304→      success,
   305→      { description }
   306→    );

   308→    if (ok) {
   309→      await load();
   310→      onChanged();
   311→    }
   312→    return ok;
   313→  }

   360→                                        {lead.phone}

   900→    const g: Record<string, TeamTaskDTO[]> = {};

   902→      (g[t.kind] ??= []).push(t);

   906→
   907→  return (
   908→    <div>
