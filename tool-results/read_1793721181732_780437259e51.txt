   131→      const detail = await api<ProjectDetailDTO>(`/api/projects/${project.id}`);

   209→          <div className="mb-6 flex gap-2 overflow-x-auto vasati-scroll pb-1 lg:hidden">

   373→function BuilderProjectsTab({
   374→  refreshKey,
   375→  onEdit,
   376→  onCreate,
   377→  editLoading,
   378→}: {
   379→  refreshKey: number;
   380→  onEdit: (project: ProjectCardDTO) => void;
   381→  onCreate: () => void;
   382→  editLoading: boolean;
   383→}) {
   384→  const navigate = useAppStore((s) => s.navigate);
   385→  const { data, loading, error, refetch } = useApiData<Paginated<ProjectCardDTO>>(
   386→    () => api("/api/projects?scope=mine&pageSize=24"),
   387→    [refreshKey]
   388→  );
   389→  const [deleting, setDeleting] = useState<ProjectCardDTO | null>(null);
   390→  const [deleteBusy, setDeleteBusy] = useState(false);
   391→
   392→  async function confirmDelete() {
   393→    if (!deleting) return;
   394→    setDeleteBusy(true);
   395→    await runAction(
   396→      () => api(`/api/projects/${deleting.id}`, { method: "DELETE" }),
   397→      "Project deleted",
   398→      { description: deleting.name, onDone: refetch }
   399→    );
   400→    setDeleteBusy(false);
   401→    setDeleting(null);
   402→  }
   403→
   404→  const projects = data?.data ?? [];
   405→
   406→  return (
   407→    <div>
   408→      <TabIntro
   409→        eyebrow="Projects"
   410→        title="Your projects"
   411→        description="Create, edit and track every project you are developing."
   412→        actions={
   413→          <Button onClick={onCreate} disabled={editLoading}>
   414→            {editLoading ? (
   415→              <Loader2 className="size-4 animate-spin" aria-hidden="true" />
   416→            ) : (
   417→              <Plus className="size-4" aria-hidden="true" />
   418→            )}
   419→            Create project
   420→          </Button>
   421→        }
   422→      />
   423→
   424→      {loading ? (
   425→        <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-3">
   426→          {[...Array(6)].map((_, i) => (
   427→            <Skeleton key={i} className="h-80 rounded-xl" />
   428→          ))}
   429→        </div>
   430→      ) : error ? (
   431→        <AdminCard>
   432→          <LoadError message={error} onRetry={refetch} />
   433→        </AdminCard>
   434→      ) : projects.length === 0 ? (
   435→        <EmptyState
   436→          icon={<Building2 className="size-6" aria-hidden="true" />}
   437→          title="No projects yet"
   438→          description="Create your first project to start listing units on the Vasati marketplace."
   439→          action={
   440→            <Button onClick={onCreate}>
   441→              <Plus className="size-4" aria-hidden="true" />
   442→              Create project
   443→            </Button>
   444→          }
   445→        />
   446→      ) : (
   447→        <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-3">
   448→          {projects.map((p) => (
   449→            <article
   450→              key={p.id}
   451→              className="flex flex-col overflow-hidden rounded-xl border bg-card shadow-sm transition-all duration-300 hover:-translate-y-1 hover:shadow-lg"
   452→            >
   453→              <div className="relative aspect-[16/10] bg-muted">
   454→                {p.mainImage ? (
   455→                  <img
   456→                    src={p.mainImage}
   457→                    alt={`${p.name} — ${p.locality}`}
   458→                    loading="lazy"
   459→                    className="size-full object-cover"
   460→                  />
   461→                ) : (
   462→                  <div className="grid size-full place-items-center text-muted-foreground">
   463→                    <Building2 className="size-8" aria-hidden="true" />
   464→                  </div>
   465→                )}
   466→                <div className="absolute left-3 top-3 flex flex-wrap gap-1.5">
   467→                  {p.featured ? (
   468→                    <span className="rounded-full bg-amber-500/95 px-2.5 py-0.5 text-[11px] font-semibold text-white shadow-sm">
   469→                      Featured
   470→                    </span>
   471→                  ) : null}
   472→                </div>
   473→                <div className="absolute right-3 top-3">
   474→                  <StatusBadge status={p.status} />
   475→                </div>
   476→              </div>
   477→              <div className="flex flex-1 flex-col p-4 sm:p-5">
   478→                <h3 className="font-display text-lg font-semibold tracking-tight text-foreground">
   479→                  {p.name}
   480→                </h3>
   481→                <p className="mt-0.5 text-[13px] text-muted-foreground">
   482→                  {p.locality}, {p.city}
   483→                </p>
   484→                <p className="mt-2 text-[13px] text-muted-foreground">
   485→                  {p.configuration ?? p.propertyType} ·{" "}
   486→                  {PROJECT_STATUSES.find((s) => s.value === p.construction)?.label ?? p.construction}
   487→                  {p.possessionDate ? ` · Possession ${p.possessionDate}` : ""}
   488→                </p>
   489→                <div className="mt-3 flex flex-wrap items-center justify-between gap-2">
   490→                  <p className="font-display text-base font-semibold text-foreground">
   491→                    {formatPriceRange(p.priceMin, p.priceMax)}
   492→                  </p>
   493→                  <p className="text-xs text-muted-foreground">
   494→                    {formatNumber(p.availableUnits)} of {formatNumber(p.totalUnits)} units available
   495→                  </p>
   496→                </div>
   497→                <div className="mt-4 flex flex-wrap gap-2 border-t pt-3.5">
   498→                  {p.status === "PUBLISHED" ? (
   499→                    <Button
   500→                      size="sm"
   501→                      variant="outline"
   502→                      onClick={() => navigate(`/projects/${p.slug}`)}
   503→                    >
   504→                      <Eye className="size-4" aria-hidden="true" />
   505→                      View
   506→                    </Button>
   507→                  ) : null}
   508→                  <Button size="sm" variant="outline" onClick={() => onEdit(p)}>
   509→                    <Pencil className="size-4" aria-hidden="true" />
   510→                    Edit
   511→                  </Button>
   512→                  <Button
   513→                    size="sm"
   514→                    variant="ghost"
   515→                    className="text-rose-600 hover:text-rose-700"
   516→                    onClick={() => setDeleting(p)}
   517→                    aria-label={`Delete ${p.name}`}
   518→                  >
   519→                    <Trash2 className="size-4" aria-hidden="true" />
   520→                    Delete
   521→                  </Button>
   522→                </div>
   523→              </div>
   524→            </article>
   525→          ))}
   526→        </div>
   527→      )}
   528→
   529→      <AlertDialog open={!!deleting} onOpenChange={(open) => !open && setDeleting(null)}>
   530→        <AlertDialogContent>
   531→          <AlertDialogHeader>
   532→            <AlertDialogTitle>Delete this project?</AlertDialogTitle>
   533→            <AlertDialogDescription>
   534→              {deleting
   535→                ? `"${deleting.name}" will be permanently removed. Projects that still have units cannot be deleted.`
   536→                : ""}
   537→            </AlertDialogDescription>
   538→          </AlertDialogHeader>
   539→          <AlertDialogFooter>
   540→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   541→            <AlertDialogAction
   542→              onClick={(e) => {
   543→                e.preventDefault();
   544→                void confirmDelete();
   545→              }}
   546→              className="bg-destructive text-white hover:bg-destructive/90"
   547→            >
   548→              {deleteBusy ? <Loader2 className="size-4 animate-spin" aria-hidden="true" /> : null}
   549→              Delete project
   550→            </AlertDialogAction>
   551→          </AlertDialogFooter>
   552→        </AlertDialogContent>
   553→      </AlertDialog>
   554→    </div>
   555→  );
   556→}

   562→function BuilderInventoryTab() {
   563→  const [page, setPage] = useState(1);
   564→  const { data, loading, error, refetch } = useApiData<Paginated<PropertyCardDTO>>(
   565→    () => api(`/api/properties?scope=mine&pageSize=48&page=${page}`),
   566→    [page]
   567→  );
   568→  const [unitForm, setUnitForm] = useState<{ open: boolean; propertyId: string | null }>({
   569→    open: false,
   570→    propertyId: null,
   571→  });
   572→  const [busyId, setBusyId] = useState<string | null>(null);
   573→
   574→  async function setAvailability(property: PropertyCardDTO, availability: string) {
   575→    setBusyId(property.id);
   576→    await runAction(
   577→      () =>
   578→        api(`/api/properties/${property.id}`, {
   579→          method: "PATCH",
   580→          body: JSON.stringify({ availability }),
   581→        }),
   582→      availability === "SOLD" ? "Unit marked as sold" : "Unit marked as available",
   583→      { description: property.title, onDone: refetch }
   584→    );
   585→    setBusyId(null);
   586→  }
   587→
   588→  const units = data?.data ?? [];
   589→
   590→  return (
   591→    <div>
   592→      <TabIntro
   593→        eyebrow="Inventory"
   594→        title="Unit inventory"
   595→        description="Every unit you have listed, including individual apartments, villas and plots."
   596→        actions={
   597→          <Button onClick={() => setUnitForm({ open: true, propertyId: null })}>
   598→            <Plus className="size-4" aria-hidden="true" />
   599→            Add unit
