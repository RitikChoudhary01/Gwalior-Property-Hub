    49→export function qs(params: Record<string, string | number | undefined | null | false>): string {
    50→  const sp = new URLSearchParams();
    51→  for (const [key, value] of Object.entries(params)) {
    52→    if (value === undefined || value === null || value === "" || value === false) continue;
    53→    sp.set(key, String(value));
    54→  }
    55→  const s = sp.toString();
    56→  return s ? `?${s}` : "";
    57→}

    63→export function useApiData<T>(
    64→  fetcher: () => Promise<T>,
    65→  deps: ReadonlyArray<unknown>
    66→): { data: T | null; loading: boolean; error: string | null; refetch: () => void } {
    67→  const [data, setData] = useState<T | null>(null);
    68→  const [loading, setLoading] = useState(true);
    69→  const [error, setError] = useState<string | null>(null);
    70→  const [tick, setTick] = useState(0);
    71→  const refetch = useCallback(() => setTick((t) => t + 1), []);
    72→
    73→  useEffect(() => {
    74→    let cancelled = false;
    75→    const run = async () => {
    76→      setLoading(true);
    77→      setError(null);
    78→      try {
    79→        const res = await fetcher();
    80→        if (!cancelled) {
    81→          setData(res);
    82→          setLoading(false);
    83→        }
    84→      } catch (err) {
    85→        if (!cancelled) {
    86→          setError(err instanceof ApiError ? err.message : "Something went wrong. Please try again.");
    87→          setLoading(false);
    88→        }
    89→      }
    90→    };
    91→    void run();
    92→    return () => {
    93→      cancelled = true;
    94→    };
    95→  }, [...deps, tick]);
    96→
    97→  return { data, loading, error, refetch };
    98→}

   100→export function useDebounced<T>(value: T, delay = 350): T {
   101→  const [debounced, setDebounced] = useState(value);
   102→  useEffect(() => {
   103→    const t = setTimeout(() => setDebounced(value), delay);
   104→    return () => clearTimeout(t);
   105→  }, [value, delay]);
   106→  return debounced;
   107→}

   109→/** Tab state derived from /dashboard?tab=x — the URL is the source of truth. */
   110→export function useDashboardTab(
   111→  validTabs: readonly string[],
   112→  initialTab?: string
   113→): [string, (tab: string) => void] {
   114→  const navigate = useAppStore((s) => s.navigate);
   115→  const route = useAppStore((s) => s.route);
   116→  const routeTab = route.query.get("tab") ?? "";
   117→  const tab =
   118→    routeTab && validTabs.includes(routeTab)
   119→      ? routeTab
   120→      : !routeTab && initialTab && validTabs.includes(initialTab)
   121→        ? initialTab
   122→        : validTabs[0];
   123→
   124→  const select = useCallback(
   125→    (next: string) => {
   126→      if (!validTabs.includes(next) || next === tab) return;
   127→      navigate(`/dashboard?tab=${next}`, { replace: true });
   128→    },
   129→    [tab, navigate, validTabs]
   130→  );
   131→
   132→  return [tab, select];
   133→}

   139→export async function runAction(
   140→  fn: () => Promise<unknown>,
   141→  success: string,
   142→  opts?: { description?: string; onDone?: () => void }
   143→): Promise<boolean> {
   144→  try {
   145→    await fn();

   150→    toast.error(err instanceof ApiError ? err.message : "Something went wrong. Please try again.");
   151→    return false;

   339→export function TablePager({
   340→  pagination,
   341→  onPage,
   342→  unit = "results",
   343→}: {
   344→  pagination: Pagination | null | undefined;
   345→  onPage: (page: number) => void;
   346→  unit?: string;
   347→}) {
   348→  if (!pagination || pagination.total === 0) return null;
   349→  const { page, totalPages, total } = pagination;
   350→
   351→  if (totalPages <= 1) {
   352→    return (
   353→      <p className="border-t px-4 py-3 text-xs text-muted-foreground sm:px-5">
   354→        {formatNumber(total)} {unit}
   355→      </p>
   356→    );
   357→  }
   358→
   359→  const from = Math.max(1, page - 2);
   360→  const to = Math.min(totalPages, page + 2);
   361→  const pages: number[] = [];
   362→  for (let p = from; p <= to; p++) pages.push(p);
   363→
   364→  return (
   365→    <div className="flex flex-wrap items-center justify-between gap-3 border-t px-4 py-3 sm:px-5">
   366→      <p className="text-xs text-muted-foreground">
   367→        Page {page} of {totalPages} · {formatNumber(total)} {unit}
   368→      </p>
   369→      <div className="flex items-center gap-1">
   370→        <Button
   371→          variant="outline"
   372→          size="sm"
   373→          disabled={page <= 1}
   374→          onClick={() => onPage(page - 1)}
   375→          aria-label="Previous page"
   376→        >
   377→          <ChevronLeft className="size-4" aria-hidden="true" />
   378→          Prev
   379→        </Button>
   380→        {pages.map((p) => (
   381→          <Button
   382→            key={p}
   383→            variant={p === page ? "default" : "outline"}
   384→            size="icon"
   385→            className="size-8 text-xs"
   386→            onClick={() => onPage(p)}
   387→            aria-label={`Page ${p}`}
   388→            aria-current={p === page ? "page" : undefined}
   389→          >
   390→            {p}
   391→          </Button>
   392→        ))}
   393→        <Button
   394→          variant="outline"
   395→          size="sm"
   396→          disabled={page >= totalPages}
   397→          onClick={() => onPage(page + 1)}
   398→          aria-label="Next page"
   399→        >
   400→          Next
   401→          <ChevronRight className="size-4" aria-hidden="true" />
   402→        </Button>
   403→      </div>
   404→    </div>
   405→  );
   406→}

   505→export function RescheduleVisitDialog({
   506→  visit,
   507→  onClose,
   508→  onDone,
   509→}: {
   510→  visit: SiteVisitDTO;
   511→  onClose: () => void;
   512→  onDone: () => void;
   513→}) {
   514→  const [date, setDate] = useState(visit.preferredDate ? visit.preferredDate.slice(0, 10) : "");
   515→  const [time, setTime] = useState(visit.preferredTime ?? "");
   516→  const [busy, setBusy] = useState(false);
   517→  const [error, setError] = useState<string | null>(null);
   518→
   519→  async function submit() {
   520→    if (!date) {
   521→      setError("Pick a new date for the visit.");
   522→      return;
   523→    }
   524→    setBusy(true);
   525→    setError(null);
   526→    try {
   527→      await api(`/api/site-visits/${visit.id}`, {
   528→        method: "PATCH",
   529→        body: JSON.stringify({
   530→          status: "RESCHEDULED",
   531→          preferredDate: date,
   532→          preferredTime: time || undefined,
   533→        }),
   534→      });
   535→      toast.success("Visit rescheduled", {
   536→        description: `${visit.property.title} — ${formatDate(date)}${time ? ` at ${time}` : ""}`,
   537→      });
   538→      onClose();
   539→      onDone();
   540→    } catch (err) {
   541→      setError(err instanceof ApiError ? err.message : "Could not reschedule the visit.");
   542→    } finally {
   543→      setBusy(false);
   544→    }
   545→  }
   546→
   547→  return (
   548→    <Dialog open onOpenChange={(open) => !open && onClose()}>
   549→      <DialogContent className="sm:max-w-md">
   550→        <DialogHeader>
   551→          <DialogTitle>Reschedule site visit</DialogTitle>
   552→          <DialogDescription>
   553→            {visit.property.title} — currently {formatDate(visit.preferredDate)} at{" "}
   554→            {visit.preferredTime}
   555→          </DialogDescription>
   556→        </DialogHeader>
   557→        <div className="grid gap-4">
   558→          <div className="grid gap-2">
   559→            <Label htmlFor="visit-date">New date</Label>
   560→            <Input
   561→              id="visit-date"
   562→              type="date"
   563→              value={date}
   564→              onChange={(e) => {
   565→                setDate(e.target.value);
   566→                setError(null);
   567→              }}
   568→            />
   569→          </div>
   570→          <div className="grid gap-2">
   571→            <Label htmlFor="visit-time">New time slot</Label>
   572→            <Select value={time} onValueChange={setTime}>
   573→              <SelectTrigger id="visit-time" className="w-full">
   574→                <SelectValue placeholder="Select a time slot" />
   575→              </SelectTrigger>
   576→              <SelectContent>
   577→                {VISIT_TIME_SLOTS.map((slot) => (
   578→                  <SelectItem key={slot} value={slot}>
   579→                    {slot}
   580→                  </SelectItem>
   581→                ))}
   582→              </SelectContent>
   583→            </Select>
   584→          </div>
   585→          {error ? <p className="text-sm text-destructive">{error}</p> : null}
   586→        </div>
   587→        <DialogFooter>
   588→          <Button variant="outline" onClick={onClose}>
   589→            Cancel
   590→          </Button>
   591→          <Button onClick={submit} disabled={busy}>
   592→            <CalendarClock className="size-4" aria-hidden="true" />
   593→            Reschedule visit
   594→          </Button>
   595→        </DialogFooter>
   596→      </DialogContent>
   597→    </Dialog>
   598→  );
   599→}

   630→      <dl className="grid grid-cols-2 gap-x-4 gap-y-2.5 border-t px-4 py-3 text-xs">
   631→        <div>
   632→          <dt className="text-muted-foreground">Visitor</dt>
   633→          <dd className="mt-0.5 font-medium text-foreground">{visit.contactName}</dd>
   634→        </div>
   635→        <div>
   636→          <dt className="text-muted-foreground">Contact</dt>
   637→          <dd className="mt-0.5 font-medium">
   638→            <a href={`tel:${visit.contactPhone}`} className="text-primary hover:underline">
   639→              {visit.contactPhone}
   640→            </a>
   641→          </dd>
   642→        </div>
   643→        <div>
   644→          <dt className="text-muted-foreground">Date</dt>
   645→          <dd className="mt-0.5 font-medium text-foreground">{formatDate(visit.preferredDate)}</dd>
   646→        </div>
   647→        <div>
   648→          <dt className="text-muted-foreground">Time</dt>
   649→          <dd className="mt-0.5 font-medium text-foreground">{visit.preferredTime}</dd>
   650→        </div>
   651→        {visit.notes ? (
   652→          <div className="col-span-2">
   653→            <dt className="text-muted-foreground">Notes</dt>
   654→            <dd className="mt-0.5 text-foreground">{visit.notes}</dd>
   655→          </div>
   656→        ) : null}
   657→      </dl>
   658→      {canConfirm || canReschedule || canComplete || canCancel ? (
   659→        <div className="mt-auto flex flex-wrap gap-2 border-t px-4 py-3">
   660→          {canConfirm ? (
   661→            <Button
   662→              size="sm"
   663→              disabled={busy}
   664→              onClick={() =>
   665→                onStatus(
   666→                  visit,
   667→                  "CONFIRMED",
   668→                  "Visit confirmed",
   669→                  `${formatDate(visit.preferredDate)} at ${visit.preferredTime}`
   670→                )
   671→              }
   672→            >
   673→              <Check className="size-4" aria-hidden="true" />
   674→              Confirm
   675→            </Button>
   676→          ) : null}
   677→          {canReschedule ? (
   678→            <Button size="sm" variant="outline" disabled={busy} onClick={() => onReschedule(visit)}>
   679→              <CalendarClock className="size-4" aria-hidden="true" />
   680→              Reschedule
   681→            </Button>
   682→          ) : null}
   683→          {canComplete ? (
   684→            <Button
   685→              size="sm"
   686→              variant="outline"
   687→              disabled={busy}
   688→              onClick={() => onStatus(visit, "COMPLETED", "Visit marked completed", visit.property.title)}
   689→            >
   690→              <CheckCircle2 className="size-4" aria-hidden="true" />
   691→              Complete
   692→            </Button>
   693→          ) : null}
   694→          {canCancel ? (
   695→            <Button
   696→              size="sm"
   697→              variant="ghost"
   698→              disabled={busy}
   699→              className="text-rose-600 hover:text-rose-700"
   700→              onClick={() => onStatus(visit, "CANCELLED", "Visit cancelled", visit.property.title)}
   701→            >
   702→              <X className="size-4" aria-hidden="true" />
   703→              Cancel
   704→            </Button>
   705→          ) : null}
   706→        </div>
   707→      ) : null}
   708→    </article>
   709→  );
   710→}

   712→export function VisitsBoard({
   713→  title = "Site visits",
   714→  description,
   715→}: {
   716→  title?: string;
   717→  description?: string;
   718→}) {
   719→  const [status, setStatus] = useState("ALL");
   720→  const [page, setPage] = useState(1);
   721→  const [busyId, setBusyId] = useState<string | null>(null);
   722→  const [rescheduling, setRescheduling] = useState<SiteVisitDTO | null>(null);
   723→
   724→  const { data, loading, error, refetch } = useApiData<{ data: SiteVisitDTO[]; pagination?: Pagination }>(
   725→    () => api(`/api/site-visits${qs({ status: status !== "ALL" ? status : null, page, pageSize: 12 })}`),
   726→    [status, page]
   727→  );
   728→
   729→  function onStatusChange(next: string) {
   730→    setStatus(next);
   731→    setPage(1);
   732→  }
   733→
   734→  async function onStatus(
   735→    visit: SiteVisitDTO,
   736→    nextStatus: string,
   737→    success: string,
   738→    description?: string
   739→  ) {
   740→    setBusyId(visit.id);
   741→    await runAction(
   742→      () =>
   743→        api(`/api/site-visits/${visit.id}`, {
   744→          method: "PATCH",
   745→          body: JSON.stringify({ status: nextStatus }),
   746→        }),
   747→      success,
   748→      { description, onDone: refetch }
   749→    );
   750→    setBusyId(null);
   751→  }
   752→
   753→  const visits = data?.data ?? [];
   754→
   755→  return (
   756→    <div className="space-y-4">
   757→      <TabIntro eyebrow="Site visits" title={title} description={description} />
   758→      <ChipBar
   759→        value={status}
   760→        onChange={onStatusChange}
   761→        options={[
   762→          { value: "ALL", label: "All" },
   763→          ...VISIT_STATUSES.map((s) => ({ value: s.value, label: s.label })),
   764→        ]}
   765→      />
   766→      {loading ? (
   767→        <SkeletonCards count={6} />
   768→      ) : error ? (
   769→        <AdminCard>
   770→          <LoadError message={error} onRetry={refetch} />
   771→        </AdminCard>
   772→      ) : visits.length === 0 ? (
   773→        <EmptyState
   774→          icon={<CalendarDays className="size-6" aria-hidden="true" />}
   775→          title="No site visits found"
   776→          description={
   777→            status === "ALL"
   778→              ? "When buyers request a property visit, it will appear here for you to coordinate."
   779→              : `No visits with status "${VISIT_STATUSES.find((s) => s.value === status)?.label ?? status}".`
   780→          }
   781→        />
   782→      ) : (
   783→        <div className="space-y-4">
   784→          <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-3">
   785→            {visits.map((visit) => (
   786→              <VisitCard
   787→                key={visit.id}
   788→                visit={visit}
   789→                busy={busyId === visit.id}
   790→                onStatus={onStatus}
   791→                onReschedule={setRescheduling}
   792→              />
   793→            ))}
   794→          </div>
   795→          <AdminCard>
   796→            <TablePager pagination={data?.pagination} onPage={setPage} unit="visits" />
   797→          </AdminCard>
   798→        </div>
   799→      )}
   800→      {rescheduling ? (
   801→        <RescheduleVisitDialog
   802→          key={rescheduling.id}
   803→          visit={rescheduling}
   804→          onClose={() => setRescheduling(null)}
   805→          onDone={refetch}
   806→        />
   807→      ) : null}
   808→    </div>
   809→  );
   810→}
