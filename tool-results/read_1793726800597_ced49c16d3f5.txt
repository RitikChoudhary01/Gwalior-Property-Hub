    63→function useApiData<T>(path: string) {
    64→  const [data, setData] = useState<T | null>(null);
    65→  const [loading, setLoading] = useState(true);
    66→  const [error, setError] = useState<string | null>(null);
    67→  const [reloadKey, setReloadKey] = useState(0);
    68→  const refetch = useCallback(() => setReloadKey((k) => k + 1), []);
    69→
    70→  // Fetches on mount and on manual refetch. All state updates happen in
    71→  // async callbacks — previously loaded data stays visible during a refresh.
    72→  useEffect(() => {
    73→    let cancelled = false;
    74→    api<T>(path)
    75→      .then((res) => {
    76→        if (cancelled) return;
    77→        setData(res);
    78→        setError(null);
    79→        setLoading(false);
    80→      })
    81→      .catch((e: unknown) => {
    82→        if (cancelled) return;
    83→        setError(e instanceof Error ? e.message : "Something went wrong. Please try again.");
    84→        setLoading(false);
    85→      });
    86→    return () => {
    87→      cancelled = true;
    88→    };
    89→  }, [path, reloadKey]);
    90→
    91→  return { data, loading, error, refetch };
    92→}

    94→function ErrorState({ message, onRetry }: { message: string; onRetry: () => void }) {
    95→  return (
    96→    <EmptyState
    97→      icon={<XCircle className="size-6" aria-hidden="true" />}
    98→      title="Could not load this section"
    99→      description={message}
   100→      action={<Button onClick={onRetry}>Try again</Button>}
   101→    />
   102→  );
   103→}

   347→const ACTIVE_VISIT_STATUSES = ["REQUESTED", "CONFIRMED", "RESCHEDULED"];
   348→
   349→function VisitsTab() {
   350→  const navigate = useAppStore((s) => s.navigate);
   351→  const { data, loading, error, refetch } = useApiData<Paginated<SiteVisitDTO>>(
   352→    "/api/site-visits?pageSize=50"
   353→  );
   354→  const [busyId, setBusyId] = useState<string | null>(null);
   355→  const [confirmVisit, setConfirmVisit] = useState<SiteVisitDTO | null>(null);
   356→
   357→  const visits = data?.data ?? [];
   358→
   359→  const todayStart = new Date();
   360→  todayStart.setHours(0, 0, 0, 0);
   361→
   362→  const isUpcoming = (v: SiteVisitDTO) =>
   363→    ACTIVE_VISIT_STATUSES.includes(v.status) && new Date(v.preferredDate) >= todayStart;
   364→  const upcoming = visits.filter(isUpcoming);
   365→  const past = visits.filter((v) => !isUpcoming(v));
   366→
   367→  async function cancelVisit(visit: SiteVisitDTO) {
   368→    setBusyId(visit.id);
   369→    try {
   370→      await api(`/api/site-visits/${visit.id}`, {
   371→        method: "PATCH",
   372→        body: JSON.stringify({ status: "CANCELLED" }),
   373→      });
   374→      toast.success("Visit request cancelled", { description: visit.property.title });
   375→      setConfirmVisit(null);
   376→      refetch();
   377→    } catch (e) {
   378→      toast.error(e instanceof Error ? e.message : "Could not cancel the visit. Please try again.");
   379→    } finally {
   380→      setBusyId(null);
   381→    }
   382→  }
   383→
   384→  if (loading) {
   385→    return (
   386→      <div className="space-y-4">
   387→        {[...Array(2)].map((_, i) => (
   388→          <Skeleton key={i} className="h-40 rounded-xl" />
   389→        ))}
   390→      </div>
   391→    );
   392→  }
   393→
   394→  if (error) return <ErrorState message={error} onRetry={refetch} />;
   395→
   396→  if (visits.length === 0) {
   397→    return (
   398→      <EmptyState
   399→        icon={<CalendarDays className="size-6" aria-hidden="true" />}
   400→        title="No site visits yet"
   401→        description="Found a home you like? Request a guided visit from the listing page and track it here."
   402→        action={<Button onClick={() => navigate("/properties")}>Browse properties</Button>}
   403→      />
   404→    );
   405→  }
   406→
   407→  function VisitCard({ visit }: { visit: SiteVisitDTO }) {
   408→    const cancellable = ACTIVE_VISIT_STATUSES.includes(visit.status) && isUpcoming(visit);
   409→    return (
   410→      <article className="rounded-xl border bg-card p-4 shadow-sm sm:p-5">
   411→        <div className="flex flex-wrap items-start justify-between gap-3">
   412→          <div className="min-w-0">
   413→            <div className="flex flex-wrap items-center gap-2">
   414→              <h3 className="truncate text-[15px] font-medium text-foreground">{visit.property.title}</h3>
   415→              <StatusBadge status={visit.status} />
   416→            </div>
   417→            <p className="mt-1 text-[13px] text-muted-foreground">
   418→              {visit.property.locality}, {visit.property.city}
   419→            </p>
   420→          </div>
   421→          <Button
   422→            type="button"
   423→            variant="outline"
   424→            size="sm"
   425→            onClick={() => navigate(`/properties/${visit.property.slug}`)}
   426→          >
   427→            View listing
   428→          </Button>
   429→        </div>
   430→        <div className="mt-4 flex flex-wrap items-center gap-x-5 gap-y-1.5 border-t pt-3.5 text-[13px] text-muted-foreground">
   431→          <span className="inline-flex items-center gap-1.5">
   432→            <CalendarDays className="size-4" aria-hidden="true" />
   433→            {formatDate(visit.preferredDate)}
   434→          </span>
   435→          <span>{visit.preferredTime}</span>
   436→          {cancellable && (
   437→            <Button
   438→              type="button"
   439→              variant="outline"
   440→              size="sm"
   441→              className="ml-auto text-destructive hover:text-destructive"
   442→              disabled={busyId === visit.id}
   443→              onClick={() => setConfirmVisit(visit)}
   444→            >
   445→              {busyId === visit.id && <Loader2 className="size-4 animate-spin" aria-hidden="true" />}
   446→              Cancel request
   447→            </Button>
   448→          )}
   449→        </div>
   450→      </article>
   451→    );
   452→  }
   453→
   454→  return (
   455→    <div className="space-y-8">
   456→      <section aria-labelledby="upcoming-visits-heading">
   457→        <h2 id="upcoming-visits-heading" className="font-display text-lg font-semibold text-foreground">
   458→          Upcoming
   459→        </h2>
   460→        {upcoming.length === 0 ? (
   461→          <p className="mt-3 rounded-xl border border-dashed bg-card/60 px-4 py-6 text-center text-sm text-muted-foreground">
   462→            No upcoming visits. Request one from any listing page.
   463→          </p>
   464→        ) : (
   465→          <div className="mt-4 space-y-4">
   466→            {upcoming.map((v) => (
   467→              <VisitCard key={v.id} visit={v} />
   468→            ))}
   469→          </div>
   470→        )}
   471→      </section>
   472→
   473→      {past.length > 0 && (
   474→        <section aria-labelledby="past-visits-heading">
   475→          <h2 id="past-visits-heading" className="font-display text-lg font-semibold text-foreground">
   476→            Past
   477→          </h2>
   478→          <div className="mt-4 space-y-4">
   479→            {past.map((v) => (
   480→              <VisitCard key={v.id} visit={v} />
   481→            ))}
   482→          </div>
   483→        </section>
   484→      )}
   485→
   486→      <AlertDialog open={!!confirmVisit} onOpenChange={(o) => !o && setConfirmVisit(null)}>
   487→        <AlertDialogContent>
   488→          <AlertDialogHeader>
   489→            <AlertDialogTitle>Cancel this visit request?</AlertDialogTitle>
   490→            <AlertDialogDescription>
   491→              {confirmVisit &&
   492→                `Your visit to "${confirmVisit.property.title}" on ${formatDate(
   493→                  confirmVisit.preferredDate
   494→                )} at ${confirmVisit.preferredTime} will be cancelled.`}
   495→            </AlertDialogDescription>
   496→          </AlertDialogHeader>
   497→          <AlertDialogFooter>
   498→            <AlertDialogCancel disabled={busyId !== null}>Keep visit</AlertDialogCancel>
   499→            <AlertDialogAction
   500→              disabled={busyId !== null}
   501→              onClick={(e) => {
   502→                e.preventDefault();
   503→                if (confirmVisit) void cancelVisit(confirmVisit);
   504→              }}
   505→              className="bg-destructive text-white hover:bg-destructive/90"
   506→            >
   507→              {busyId !== null && <Loader2 className="size-4 animate-spin" aria-hidden="true" />}
   508→              Cancel visit
   509→            </AlertDialogAction>
   510→          </AlertDialogFooter>
   511→        </AlertDialogContent>
   512→      </AlertDialog>
   513→    </div>
   514→  );
   515→}
