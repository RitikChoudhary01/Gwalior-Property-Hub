    98→function useApiData<T>(path: string) {
    99→  const [data, setData] = useState<T | null>(null);
   100→  const [loading, setLoading] = useState(true);
   101→  const [error, setError] = useState<string | null>(null);
   102→  const [reloadKey, setReloadKey] = useState(0);
   103→  const refetch = useCallback(() => setReloadKey((k) => k + 1), []);
   104→
   105→  // Fetches on mount and on manual refetch. All state updates happen in
   106→  // async callbacks — previously loaded data stays visible during a refresh.
   107→  useEffect(() => {
   108→    let cancelled = false;
   109→    api<T>(path)
   110→      .then((res) => {
   111→        if (cancelled) return;
   112→        setData(res);
   113→        setError(null);
   114→        setLoading(false);
   115→      })
   116→      .catch((e: unknown) => {
   117→        if (cancelled) return;
   118→        setError(e instanceof Error ? e.message : "Something went wrong. Please try again.");
   119→        setLoading(false);
   120→      });
   121→    return () => {
   122→      cancelled = true;
   123→    };
   124→  }, [path, reloadKey]);
   125→
   126→  return { data, loading, error, refetch };
   127→}

   129→function ErrorState({ message, onRetry }: { message: string; onRetry: () => void }) {
   130→  return (
   131→    <EmptyState
   132→      icon={<XCircle className="size-6" aria-hidden="true" />}
   133→      title="Could not load this section"
   134→      description={message}
   135→      action={<Button onClick={onRetry}>Try again</Button>}
   136→    />
   137→  );
   138→}

   576→  const isUpcoming = (v: SiteVisitDTO) =>
   577→    ACTIVE_VISIT_STATUSES.includes(v.status) && new Date(v.preferredDate) >= todayStart;
   578→  const upcoming = visits.filter(isUpcoming);
   579→  const past = visits.filter((v) => !isUpcoming(v));
   580→
   581→  async function cancelVisit(visit: SiteVisitDTO) {
   582→    setBusyId(visit.id);
   583→    try {
   584→      await api(`/api/site-visits/${visit.id}`, {
   585→        method: "PATCH",
   586→        body: JSON.stringify({ status: "CANCELLED" }),
   587→      });
   588→      toast.success("Visit request cancelled", { description: visit.property.title });
   589→      setConfirmVisit(null);
   590→      refetch();
   591→    } catch (e) {
   592→      toast.error(e instanceof Error ? e.message : "Could not cancel the visit. Please try again.");
   593→    } finally {
   594→      setBusyId(null);
   595→    }
   596→  }
   597→
   598→  if (loading) {
   599→    return (
   600→      <div className="space-y-4">
   601→        {[...Array(2)].map((_, i) => (
   602→          <Skeleton key={i} className="h-40 rounded-xl" />
   603→        ))}
   604→      </div>
   605→    );
   606→  }
   607→
   608→  if (error) return <ErrorState message={error} onRetry={refetch} />;
   609→
   610→  if (visits.length === 0) {
   611→    return (
   612→      <EmptyState
   613→        icon={<CalendarDays className="size-6" aria-hidden="true" />}
   614→        title="No site visits yet"
   615→        description="Found a home you like? Request a guided visit from the listing page and track it here."
   616→        action={<Button onClick={() => navigate("/properties")}>Browse properties</Button>}
   617→      />
   618→    );
   619→  }
   620→
   621→  function VisitCard({ visit }: { visit: SiteVisitDTO }) {
   622→    const cancellable = ACTIVE_VISIT_STATUSES.includes(visit.status) && isUpcoming(visit);
   623→    return (
   624→      <article className="rounded-xl border bg-card p-4 shadow-sm sm:p-5">
   625→        <div className="flex flex-wrap items-start justify-between gap-3">
   626→          <div className="min-w-0">
   627→            <div className="flex flex-wrap items-center gap-2">
   628→              <h3 className="truncate text-[15px] font-medium text-foreground">{visit.property.title}</h3>
   629→              <StatusBadge status={visit.status} />
   630→            </div>
   631→            <p className="mt-1 text-[13px] text-muted-foreground">
   632→              {visit.property.locality}, {visit.property.city}
   633→            </p>
   634→          </div>
   635→          <Button
   636→            type="button"
   637→            variant="outline"
   638→            size="sm"
   639→            onClick={() => navigate(`/properties/${visit.property.slug}`)}
   640→          >
   641→            View listing
   642→          </Button>
   643→        </div>
   644→        <div className="mt-4 flex flex-wrap items-center gap-x-5 gap-y-1.5 border-t pt-3.5 text-[13px] text-muted-foreground">
   645→          <span className="inline-flex items-center gap-1.5">
   646→            <CalendarDays className="size-4" aria-hidden="true" />
   647→            {formatDate(visit.preferredDate)}
   648→          </span>
   649→          <span>{visit.preferredTime}</span>
   650→          {cancellable && (
   651→            <Button
   652→              type="button"
   653→              variant="outline"
   654→              size="sm"
   655→              className="ml-auto text-destructive hover:text-destructive"
   656→              disabled={busyId === visit.id}
   657→              onClick={() => setConfirmVisit(visit)}
   658→            >
   659→              {busyId === visit.id && <Loader2 className="size-4 animate-spin" aria-hidden="true" />}
   660→              Cancel request
   661→            </Button>
   662→          )}
   663→        </div>
   664→      </article>
   665→    );
   666→  }
   667→
   668→  return (
   669→    <div className="space-y-8">
   670→      <section aria-labelledby="upcoming-visits-heading">
   671→        <h2 id="upcoming-visits-heading" className="font-display text-lg font-semibold text-foreground">
   672→          Upcoming
   673→        </h2>
   674→        {upcoming.length === 0 ? (
   675→          <p className="mt-3 rounded-xl border border-dashed bg-card/60 px-4 py-6 text-center text-sm text-muted-foreground">
   676→            No upcoming visits. Request one from any listing page.
   677→          </p>
   678→        ) : (
   679→          <div className="mt-4 space-y-4">
   680→            {upcoming.map((v) => (
   681→              <VisitCard key={v.id} visit={v} />
   682→            ))}
   683→          </div>
   684→        )}
   685→      </section>
   686→
   687→      {past.length > 0 && (
   688→        <section aria-labelledby="past-visits-heading">
   689→          <h2 id="past-visits-heading" className="font-display text-lg font-semibold text-foreground">
   690→            Past
   691→          </h2>
   692→          <div className="mt-4 space-y-4">
   693→            {past.map((v) => (
   694→              <VisitCard key={v.id} visit={v} />
   695→            ))}
   696→          </div>
   697→        </section>
   698→      )}
   699→
   700→      <AlertDialog open={!!confirmVisit} onOpenChange={(o) => !o && setConfirmVisit(null)}>
   701→        <AlertDialogContent>
   702→          <AlertDialogHeader>
   703→            <AlertDialogTitle>Cancel this visit request?</AlertDialogTitle>
   704→            <AlertDialogDescription>
   705→              {confirmVisit &&
   706→                `Your visit to "${confirmVisit.property.title}" on ${formatDate(
   707→                  confirmVisit.preferredDate
   708→                )} at ${confirmVisit.preferredTime} will be cancelled.`}
   709→            </AlertDialogDescription>
   710→          </AlertDialogHeader>
   711→          <AlertDialogFooter>
   712→            <AlertDialogCancel disabled={busyId !== null}>Keep visit</AlertDialogCancel>
   713→            <AlertDialogAction
   714→              disabled={busyId !== null}
   715→              onClick={(e) => {
   716→                e.preventDefault();
   717→                if (confirmVisit) void cancelVisit(confirmVisit);
   718→              }}
   719→              className="bg-destructive text-white hover:bg-destructive/90"
   720→            >
   721→              {busyId !== null && <Loader2 className="size-4 animate-spin" aria-hidden="true" />}
   722→              Cancel visit
   723→            </AlertDialogAction>
   724→          </AlertDialogFooter>
   725→        </AlertDialogContent>
   726→      </AlertDialog>
   727→    </div>
   728→  );
   729→}
