   155→function isValidArea(value: string): boolean {
   156→  const n = Number(value.trim());
   157→  return Number.isFinite(n) && Number.isInteger(n) && n >= 100;
   158→}
   159→
   160→function isValidCoord(value: string, range: number): boolean {
   161→  const n = Number(value.trim());
   162→  return Number.isFinite(n) && n >= -range && n <= range;
   163→}

   244→  // ---------- photos ----------

   246→  async function handleFiles(fileList: FileList | File[]) {
   247→    const files = Array.from(fileList);
   248→    let attached = images.length + uploads.length;
   249→
   250→    for (const file of files) {
   251→      if (attached >= MAX_PHOTOS) {
   252→        toast.error(`You can upload up to ${MAX_PHOTOS} photos.`);
   253→        break;
   254→      }
   255→      if (!ACCEPTED_TYPES.includes(file.type)) {
   256→        toast.error(`${file.name}: only JPEG, PNG or WebP photos are allowed.`);
   257→        continue;
   258→      }
   259→      if (file.size > MAX_FILE_MB * 1024 * 1024) {
   260→        toast.error(`${file.name}: photos must be ${MAX_FILE_MB} MB or smaller.`);
   261→        continue;
   262→      }
   263→
   264→      attached += 1;
   265→      const id = `${file.name}-${Date.now()}-${Math.random().toString(36).slice(2)}`;
   266→      setUploads((u) => [...u, { id, name: file.name, progress: 0 }]);
   267→      try {
   268→        const res = await uploadFile(file, (pct) => {
   269→          setUploads((u) => u.map((x) => (x.id === id ? { ...x, progress: pct } : x)));
   270→        });
   271→        setImages((imgs) => [...imgs, { url: res.url }]);
   272→        setUploads((u) => u.filter((x) => x.id !== id));
   273→      } catch {
   274→        setUploads((u) => u.filter((x) => x.id !== id));
   275→        toast.error(`${file.name}: upload failed. Please try again.`);
   276→      }
   277→    }
   278→  }
   279→
   280→  function moveImage(index: number, dir: -1 | 1) {
   281→    const target = index + dir;
   282→    if (target < 0 || target >= images.length) return;
   283→    setImages((prev) => {
   284→      const next = [...prev];
   285→      [next[index], next[target]] = [next[target], next[index]];
   286→      return next;
   287→    });
   288→    setMainIndex((prev) => {
   289→      if (prev === index) return target;
   290→      if (prev === target) return index;
   291→      return prev;
   292→    });
   293→  }
   294→
   295→  function removeImage(index: number) {
   296→    setImages((prev) => prev.filter((_, i) => i !== index));
   297→    setMainIndex((prev) => {
   298→      if (images.length <= 1) return 0;
   299→      if (index < prev) return prev - 1;
   300→      if (index === prev) return Math.min(prev, images.length - 2);
   301→      return prev;
   302→    });
   303→  }

   305→  // ---------- validation + submit ----------

   307→  function validate(): Record<string, string> {
   308→    const errs: Record<string, string> = {};
   309→    if (form.title.trim().length < 10) errs.title = "Title must be at least 10 characters.";
   310→    if (descriptionLength < 50) errs.description = "Description must be at least 50 characters.";
   311→    const price = Number(form.price.trim());
   312→    if (form.price.trim() === "" || !Number.isFinite(price) || price < 500000) {
   313→      errs.price = "Enter a price of at least ₹5,00,000.";
   314→    } else if (!Number.isInteger(price)) {
   315→      errs.price = "Price must be a whole number in rupees.";
   316→    }
   317→    if (!form.city) errs.city = "Select a city.";
   318→    if (!form.locality) errs.locality = "Select a locality.";
   319→    if (form.address.trim().length < 10) errs.address = "Enter the full address (at least 10 characters).";
   320→    if (!isPlot && (form.bhk === "" || form.bhk === "na")) {
   321→      errs.bhk = "Select the number of bedrooms.";
   322→    }
   323→    if (images.length === 0) {
   324→      errs.photos = "Add at least one photo — listings with photos get far more enquiries.";
   325→    }
   326→    if (form.carpetArea.trim() !== "" && !isValidArea(form.carpetArea)) {
   327→      errs.carpetArea = "Enter a valid carpet area of at least 100 sq.ft.";
   328→    }
   329→    if (form.builtUpArea.trim() !== "" && !isValidArea(form.builtUpArea)) {
   330→      errs.builtUpArea = "Enter a valid built-up area of at least 100 sq.ft.";
   331→    }
   332→    if (isPlot && form.plotArea.trim() !== "" && !isValidArea(form.plotArea)) {
   333→      errs.plotArea = "Enter a valid plot area of at least 100 sq.ft.";
   334→    }
   335→    if (form.latitude.trim() !== "" && !isValidCoord(form.latitude, 90)) {
   336→      errs.latitude = "Latitude must be a number between -90 and 90.";
   337→    }
   338→    if (form.longitude.trim() !== "" && !isValidCoord(form.longitude, 180)) {
   339→      errs.longitude = "Longitude must be a number between -180 and 180.";
   340→    }
   341→    return errs;
   342→  }

   344→  function buildBody(): Record<string, unknown> {
   345→    const toNum = (v: string): number | null => {
   346→      const t = v.trim();
   347→      if (t === "") return null;
   348→      const n = Number(t);
   349→      return Number.isFinite(n) ? n : null;
   350→    };
   351→    return {
   352→      title: form.title.trim(),
   353→      description: form.description.trim(),
   354→      type: form.type,
   355→      price: Number(form.price.trim()),
   356→      negotiable: form.negotiable,
   357→      city: form.city,
   358→      state: cityEntry?.state ?? "",
   359→      locality: form.locality,
   360→      address: form.address.trim(),
   361→      latitude: toNum(form.latitude),
   362→      longitude: toNum(form.longitude),
   363→      bhk: !isPlot && form.bhk !== "" && form.bhk !== "na" ? Number(form.bhk) : null,
   364→      bathrooms: form.bathrooms !== "" ? Number(form.bathrooms) : null,
   365→      carpetArea: toNum(form.carpetArea),
   366→      builtUpArea: toNum(form.builtUpArea),
   367→      plotArea: isPlot ? toNum(form.plotArea) : null,
   368→      furnishing: form.furnishing !== "" ? form.furnishing : null,
   369→      parking: form.parking === "" ? 0 : Number(form.parking),
   370→      floorInfo: form.floorInfo.trim() !== "" ? form.floorInfo.trim() : null,
   371→      facing: form.facing !== "" ? form.facing : null,
   372→      constructionStatus: form.constructionStatus,
   373→      possessionDate: form.constructionStatus === "UNDER_CONSTRUCTION" && form.possessionDate !== "" ? form.possessionDate : null,
   374→      ageOfProperty: form.constructionStatus === "READY_TO_MOVE" && form.ageOfProperty !== "" ? form.ageOfProperty : null,
   375→      reraNumber: form.reraNumber.trim() !== "" ? form.reraNumber.trim() : null,
   376→      amenities,
   377→      images: images.map((img, i) => ({ url: img.url, isMain: i === mainIndex })),
   378→    };
   379→  }

   381→  async function handleSubmit(kind: SubmitKind) {
   382→    const errs = validate();
   383→    if (Object.keys(errs).length > 0) {
   384→      setErrors(errs);
   385→      toast.error("Please fix the highlighted fields before saving.");
   386→      return;
   387→    }
   388→    setErrors({});
   389→    setApiError(null);
   390→    setSubmitting(kind);
   391→    try {
   392→      const body = buildBody();
   393→      if (property) {
   394→        if (canChooseDraft) {
   395→          body.status = kind === "submit" ? "PENDING" : "DRAFT";
   396→        }
   397→        await api(`/api/properties/${property.id}`, { method: "PATCH", body: JSON.stringify(body) });
   398→      } else {
   399→        body.draft = kind === "draft";
   400→        await api("/api/properties", { method: "POST", body: JSON.stringify(body) });
   401→      }
   402→      if (kind === "draft") {
   403→        toast.success("Draft saved", { description: "You can finish and submit it any time." });
   404→      } else if (isEdit) {
   405→        toast.success("Listing updated", { description: "It is now pending approval." });
   406→      } else {
   407→        toast.success("Listing submitted for approval", {
   408→          description: "Our team reviews new listings, usually within 24 hours.",
   409→        });
   410→      }
   411→      onSaved();
   412→      onClose();
   413→    } catch (e) {
   414→      setApiError(e instanceof Error ? e.message : "Could not save the listing. Please try again.");
   415→    } finally {
   416→      setSubmitting(null);
   417→    }
   418→  }

   426→          {isEdit ? "Edit listing" : "List a new property"}
   427→        </DialogTitle>
   428→        <DialogDescription>
   429→          {isEdit
   430→            ? "Update the details below. Your latest photos and amenities are saved with this listing."
   431→            : "Accurate details and good photos help your listing get approved and found faster."}

   479→                aria-invalid={!!errors.description}
   480→              />

   482→                <FieldError message={errors.description} />
   483→                <p
   484→                  className={cn(
   485→                    "text-xs",
   486→                    descriptionLength < 50 ? "font-medium text-destructive" : "text-muted-foreground"

   488→                >
   489→                  {descriptionLength} / 50 characters minimum
   490→                </p>

   532→                  {Number.isFinite(priceValue) && priceValue >= 500000

   535→                </p>
   536→              </div>

   539→            <div className="flex items-center justify-between gap-4 rounded-lg border bg-card p-3.5">
