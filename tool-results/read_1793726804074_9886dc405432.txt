    49→export function qs(params: Record<string, string | number | undefined | null | false>): string {
    50→  const sp = new URLSearchParams();
    51→  for (const [key, value] of Object.entries(params)) {
    52→    if (value === undefined || value === null || value === "" || value === false) continue;
    53→    sp.set(key, String(value));
    54→  }
    55→  const s = sp.toString();
    56→  return s ? `?${s}` : "";
    57→}

    63→export function useApiData<T>(
    64→  fetcher: () => Promise<T>,
    65→  deps: ReadonlyArray<unknown>
    66→): { data: T | null; loading: boolean; error: string | null; refetch: () => void } {
    67→  const [data, setData] = useState<T | null>(null);
    68→  const [loading, setLoading] = useState(true);
    69→  const [error, setError] = useState<string | null>(null);
    70→  const [tick, setTick] = useState(0);
    71→  const refetch = useCallback(() => setTick((t) => t + 1), []);
    72→
    73→  useEffect(() => {
    74→    let cancelled = false;
    75→    const run = async () => {
    76→      setLoading(true);
    77→      setError(null);
    78→      try {
    79→        const res = await fetcher();
    80→        if (!cancelled) {
    81→          setData(res);
    82→          setLoading(false);
    83→        }
    84→      } catch (err) {
    85→        if (!cancelled) {
    86→          setError(err instanceof ApiError ? err.message : "Something went wrong. Please try again.");
    87→          setLoading(false);
    88→        }
    89→      }
    90→    };
    91→    void run();
    92→    return () => {
    93→      cancelled = true;
    94→    };
    95→  }, [...deps, tick]);
    96→
    97→  return { data, loading, error, refetch };
    98→}

   100→export function useDebounced<T>(value: T, delay = 350): T {
   101→  const [debounced, setDebounced] = useState(value);
   102→  useEffect(() => {
   103→    const t = setTimeout(() => setDebounced(value), delay);
   104→    return () => clearTimeout(t);
   105→  }, [value, delay]);
   106→  return debounced;
   107→}

   109→/** Tab state derived from /dashboard?tab=x — the URL is the source of truth. */
   110→export function useDashboardTab(
   111→  validTabs: readonly string[],
   112→  initialTab?: string
   113→): [string, (tab: string) => void] {
   114→  const navigate = useAppStore((s) => s.navigate);
   115→  const route = useAppStore((s) => s.route);
   116→  const routeTab = route.query.get("tab") ?? "";
   117→  const tab =
   118→    routeTab && validTabs.includes(routeTab)
   119→      ? routeTab
   120→      : !routeTab && initialTab && validTabs.includes(initialTab)
   121→        ? initialTab
   122→        : validTabs[0];
   123→
   124→  const select = useCallback(
   125→    (next: string) => {
   126→      if (!validTabs.includes(next) || next === tab) return;
   127→      navigate(`/dashboard?tab=${next}`, { replace: true });
   128→    },
   129→    [tab, navigate, validTabs]
   130→  );
   131→
   132→  return [tab, select];
   133→}

   139→export async function runAction(
   140→  fn: () => Promise<unknown>,
   141→  success: string,
   142→  opts?: { description?: string; onDone?: () => void }
   143→): Promise<boolean> {
   144→  try {
   145→    await fn();

   149→  } catch (err) {
   150→    toast.error(err instanceof ApiError ? err.message : "Something went wrong. Please try again.");
   151→    return false;
   152→  }

   339→export function TablePager({
   340→  pagination,
   341→  onPage,
   342→  unit = "results",
   343→}: {
   344→  pagination: Pagination | null | undefined;
   345→  onPage: (page: number) => void;
   346→  unit?: string;
   347→}) {
   348→  if (!pagination || pagination.total === 0) return null;
   349→  const { page, totalPages, total } = pagination;
   350→
   351→  if (totalPages <= 1) {
   352→    return (
   353→      <p className="border-t px-4 py-3 text-xs text-muted-foreground sm:px-5">
   354→        {formatNumber(total)} {unit}
   355→      </p>
   356→    );
   357→  }
   358→
   359→  const from = Math.max(1, page - 2);
   360→  const to = Math.min(totalPages, page + 2);
   361→  const pages: number[] = [];
   362→  for (let p = from; p <= to; p++) pages.push(p);
   363→
   364→  return (
   365→    <div className="flex flex-wrap items-center justify-between gap-3 border-t px-4 py-3 sm:px-5">
   366→      <p className="text-xs text-muted-foreground">
   367→        Page {page} of {totalPages} · {formatNumber(total)} {unit}
   368→      </p>
   369→      <div className="flex items-center gap-1">
   370→        <Button
   371→          variant="outline"
   372→          size="sm"
   373→          disabled={page <= 1}
   374→          onClick={() => onPage(page - 1)}
   375→          aria-label="Previous page"
   376→        >
   377→          <ChevronLeft className="size-4" aria-hidden="true" />
   378→          Prev
   379→        </Button>
   380→        {pages.map((p) => (
   381→          <Button
   382→            key={p}
   383→            variant={p === page ? "default" : "outline"}
   384→            size="icon"
   385→            className="size-8 text-xs"
   386→            onClick={() => onPage(p)}
   387→            aria-label={`Page ${p}`}
   388→            aria-current={p === page ? "page" : undefined}
   389→          >
   390→            {p}
   391→          </Button>
   392→        ))}
   393→        <Button
   394→          variant="outline"
   395→          size="sm"
   396→          disabled={page >= totalPages}
   397→          onClick={() => onPage(page + 1)}
   398→          aria-label="Next page"
   399→        >
   400→          Next
   401→          <ChevronRight className="size-4" aria-hidden="true" />
   402→        </Button>
   403→      </div>
   404→    </div>
   405→  );
   406→}

   504→export function RescheduleVisitDialog({
   505→  visit,
   506→  onClose,
   507→  onDone,
   508→}: {
   509→  visit: SiteVisitDTO;
   510→  onClose: () => void;
   511→  onDone: () => void;
   512→}) {
   513→  const [date, setDate] = useState(visit.preferredDate ? visit.preferredDate.slice(0, 10) : "");
   514→  const [time, setTime] = useState(visit.preferredTime ?? "");
   515→  const [busy, setBusy] = useState(false);
   516→  const [error, setError] = useState<string | null>(null);
   517→
   518→  async function submit() {
   519→    if (!date) {
   520→      setError("Pick a new date for the visit.");
   521→      return;
   522→    }
   523→    setBusy(true);
   524→    setError(null);
   525→    try {
   526→      await api(`/api/site-visits/${visit.id}`, {
   527→        method: "PATCH",
   528→        body: JSON.stringify({
   529→          status: "RESCHEDULED",
   530→          preferredDate: date,
   531→          preferredTime: time || undefined,
   532→        }),
   533→      });
   534→      toast.success("Visit rescheduled", {
   535→        description: `${visit.property.title} — ${formatDate(date)}${time ? ` at ${time}` : ""}`,
   536→      });
   537→      onClose();
   538→      onDone();
   539→    } catch (err) {
   540→      setError(err instanceof ApiError ? err.message : "Could not reschedule the visit.");
   541→    } finally {
   542→      setBusy(false);
   543→    }
   544→  }
   545→
   546→  return (
   547→    <Dialog open onOpenChange={(open) => !open && onClose()}>
   548→      <DialogContent className="sm:max-w-md">
   549→        <DialogHeader>
   550→          <DialogTitle>Reschedule site visit</DialogTitle>
   551→          <DialogDescription>
   552→            {visit.property.title} — currently {formatDate(visit.preferredDate)} at{" "}
   553→            {visit.preferredTime}
   554→          </DialogDescription>
   555→        </DialogHeader>
   556→        <div className="grid gap-4">
   557→          <div className="grid gap-2">
   558→            <Label htmlFor="visit-date">New date</Label>
   559→            <Input
   560→              id="visit-date"
   561→              type="date"
   562→              value={date}
   563→              onChange={(e) => {
   564→                setDate(e.target.value);
   565→                setError(null);
   566→              }}
   567→            />
   568→          </div>
   569→          <div className="grid gap-2">
   570→            <Label htmlFor="visit-time">New time slot</Label>
   571→            <Select value={time} onValueChange={setTime}>
   572→              <SelectTrigger id="visit-time" className="w-full">
   573→                <SelectValue placeholder="Select a time slot" />
   574→              </SelectTrigger>
   575→              <SelectContent>
   576→                {VISIT_TIME_SLOTS.map((slot) => (
   577→                  <SelectItem key={slot} value={slot}>
   578→                    {slot}
   579→                  </SelectItem>
   580→                ))}
   581→              </SelectContent>
   582→            </Select>
   583→          </div>
   584→          {error ? <p className="text-sm text-destructive">{error}</p> : null}
   585→        </div>
   586→        <DialogFooter>
   587→          <Button variant="outline" onClick={onClose}>
   588→            Cancel
   589→          </Button>
   590→          <Button onClick={submit} disabled={busy}>
   591→            <CalendarClock className="size-4" aria-hidden="true" />
   592→            Reschedule visit
   593→          </Button>
   594→        </DialogFooter>
   595→      </DialogContent>
   596→    </Dialog>
   597→  );
   598→}

   629→      <dl className="grid grid-cols-2 gap-x-4 gap-y-2.5 border-t px-4 py-3 text-xs">
   630→        <div>
   631→          <dt className="text-muted-foreground">Visitor</dt>
   632→          <dd className="mt-0.5 font-medium text-foreground">{visit.contactName}</dd>
   633→        </div>
   634→        <div>
   635→          <dt className="text-muted-foreground">Contact</dt>
   636→          <dd className="mt-0.5 font-medium">
   637→            <a href={`tel:${visit.contactPhone}`} className="text-primary hover:underline">
   638→              {visit.contactPhone}
   639→            </a>
   640→          </dd>
   641→        </div>
   642→        <div>
   643→          <dt className="text-muted-foreground">Date</dt>
   644→          <dd className="mt-0.5 font-medium text-foreground">{formatDate(visit.preferredDate)}</dd>
   645→        </div>
   646→        <div>
   647→          <dt className="text-muted-foreground">Time</dt>
   648→          <dd className="mt-0.5 font-medium text-foreground">{visit.preferredTime}</dd>
   649→        </div>
   650→        {visit.notes ? (
   651→          <div className="col-span-2">
   652→            <dt className="text-muted-foreground">Notes</dt>
   653→            <dd className="mt-0.5 text-foreground">{visit.notes}</dd>
   654→          </div>
   655→        ) : null}
   656→      </dl>
   657→      {canConfirm || canReschedule || canComplete || canCancel ? (
   658→        <div className="mt-auto flex flex-wrap gap-2 border-t px-4 py-3">
   659→          {canConfirm ? (
   660→            <Button
   661→              size="sm"
   662→              disabled={busy}
   663→              onClick={() =>
   664→                onStatus(
   665→                  visit,
   666→                  "CONFIRMED",
   667→                  "Visit confirmed",
   668→                  `${formatDate(visit.preferredDate)} at ${visit.preferredTime}`
   669→                )
   670→              }
   671→            >
   672→              <Check className="size-4" aria-hidden="true" />
   673→              Confirm
   674→            </Button>
   675→          ) : null}
   676→          {canReschedule ? (
   677→            <Button size="sm" variant="outline" disabled={busy} onClick={() => onReschedule(visit)}>
   678→              <CalendarClock className="size-4" aria-hidden="true" />
   679→              Reschedule
   680→            </Button>
   681→          ) : null}
   682→          {canComplete ? (
   683→            <Button
   684→              size="sm"
   685→              variant="outline"
   686→              disabled={busy}
   687→              onClick={() => onStatus(visit, "COMPLETED", "Visit marked completed", visit.property.title)}
   688→            >
   689→              <CheckCircle2 className="size-4" aria-hidden="true" />
   690→              Complete
   691→            </Button>
   692→          ) : null}
   693→          {canCancel ? (
   694→            <Button
   695→              size="sm"
   696→              variant="ghost"
   697→              disabled={busy}
   698→              className="text-rose-600 hover:text-rose-700"
   699→              onClick={() => onStatus(visit, "CANCELLED", "Visit cancelled", visit.property.title)}
   700→            >
   701→              <X className="size-4" aria-hidden="true" />
   702→              Cancel
   703→            </Button>
   704→          ) : null}
   705→        </div>
   706→      ) : null}
   707→    </article>
   708→  );
   709→}

   711→export function VisitsBoard({
   712→  title = "Site visits",
   713→  description,
   714→}: {
   715→  title?: string;
   716→  description?: string;
   717→}) {
   718→  const [status, setStatus] = useState("ALL");
   719→  const [page, setPage] = useState(1);
   720→  const [busyId, setBusyId] = useState<string | null>(null);
   721→  const [rescheduling, setRescheduling] = useState<SiteVisitDTO | null>(null);
   722→
   723→  const { data, loading, error, refetch } = useApiData<{ data: SiteVisitDTO[]; pagination?: Pagination }>(
   724→    () => api(`/api/site-visits${qs({ status: status !== "ALL" ? status : null, page, pageSize: 12 })}`),
   725→    [status, page]
   726→  );
   727→
   728→  function onStatusChange(next: string) {
   729→    setStatus(next);
   730→    setPage(1);
   731→  }
   732→
   733→  async function onStatus(
   734→    visit: SiteVisitDTO,
   735→    nextStatus: string,
   736→    success: string,
   737→    description?: string
   738→  ) {
   739→    setBusyId(visit.id);
   740→    await runAction(
   741→      () =>
   742→        api(`/api/site-visits/${visit.id}`, {
   743→          method: "PATCH",
   744→          body: JSON.stringify({ status: nextStatus }),
   745→        }),
   746→      success,
   747→      { description, onDone: refetch }
   748→    );
   749→    setBusyId(null);
   750→  }
   751→
   752→  const visits = data?.data ?? [];
   753→
   754→  return (
   755→    <div className="space-y-4">
   756→      <TabIntro eyebrow="Site visits" title={title} description={description} />
   757→      <ChipBar
   758→        value={status}
   759→        onChange={onStatusChange}
   760→        options={[
   761→          { value: "ALL", label: "All" },
   762→          ...VISIT_STATUSES.map((s) => ({ value: s.value, label: s.label })),
   763→        ]}
   764→      />
   765→      {loading ? (
   766→        <SkeletonCards count={6} />
   767→      ) : error ? (
   768→        <AdminCard>
   769→          <LoadError message={error} onRetry={refetch} />
   770→        </AdminCard>
   771→      ) : visits.length === 0 ? (
   772→        <EmptyState
   773→          icon={<CalendarDays className="size-6" aria-hidden="true" />}
   774→          title="No site visits found"
   775→          description={
   776→            status === "ALL"
   777→              ? "When buyers request a property visit, it will appear here for you to coordinate."
   778→              : `No visits with status "${VISIT_STATUSES.find((s) => s.value === status)?.label ?? status}".`
   779→          }
   780→        />
   781→      ) : (
   782→        <div className="space-y-4">
   783→          <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-3">
   784→            {visits.map((visit) => (
   785→              <VisitCard
   786→                key={visit.id}
   787→                visit={visit}
   788→                busy={busyId === visit.id}
   789→                onStatus={onStatus}
   790→                onReschedule={setRescheduling}
   791→              />
   792→            ))}
   793→          </div>
   794→          <AdminCard>
   795→            <TablePager pagination={data?.pagination} onPage={setPage} unit="visits" />
   796→          </AdminCard>
   797→        </div>
   798→      )}
   799→      {rescheduling ? (
   800→        <RescheduleVisitDialog
   801→          key={rescheduling.id}
   802→          visit={rescheduling}
   803→          onClose={() => setRescheduling(null)}
   804→          onDone={refetch}
   805→        />
   806→      ) : null}
   807→    </div>
   808→  );
   809→}
